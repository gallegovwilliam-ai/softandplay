sudo nano /etc/systemd/system/daphne.service
[Unit]
Description=daphne daemon
After=network.target


[Service]
PIDFile=/run/daphne/pid
User=root
Group=root
WorkingDirectory=/opt/choix57/app/
ExecStart=/opt/choix57/env3/bin/daphne -b localhost -p 9000 app.asgi:application
ExecReload=/bin/kill -s HUP $MAINPID
ExecStop=/bin/kill -s TERM $MAINPID
Restart=on-abort
PrivateTmp=true


[Install]
WantedBy=multi-user.target


sudo nano /etc/systemd/system/gunicorn.service
[Unit]
Description = SampleApp
After = network.target

[Service]
PIDFile = /run/cmdb/cmdb.pids
WorkingDirectory = /opt/choix57/app
ExecStartPre = /bin/mkdir /run/cmdb
ExecStart = /opt/choix57/env3/bin/gunicorn app.wsgi:application --timeout 0 --workers 3 --bind 127.0.0.1:8000
ExecReload = /bin/kill -s HUP $MAINPID
ExecStop = /bin/kill -s TERM $MAINPID
ExecStopPost = /bin/rm -rf /run/cmdb

[Install]
WantedBy = multi-user.target

sudo nano /etc/systemd/system/rqworker.service
[Unit]
Description = rqworkers
After = network.target

[Service]
PIDFile = /run/cmdb/cmdb.pid
WorkingDirectory = /opt/choix57/app
ExecStartPre = /bin/mkdir -p /run/cmdb
ExecStart = /opt/choix57/env3/bin/python /opt/choix57/app/manage.py rqworker default
ExecReload = /bin/kill -s HUP $MAINPID
ExecStop = /bin/kill -s TERM $MAINPID
ExecStopPost = /bin/rm -rf /run/cmdb

[Install]
WantedBy = multi-user.target

sudo systemctl enable gunicorn && sudo systemctl enable daphne && sudo systemctl enable rqworker
